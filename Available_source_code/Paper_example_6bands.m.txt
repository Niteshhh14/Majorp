% Audiogram Matching for 6 standard audiograms using 10 bands of filters with fixed shifts and gain & using continuous VBW filter
fs      = 16000;
BW_val  = [200 300 400 500 600 800];  % sort(unique(BW)); %200 300 400 500 600 700 
fixedBW = BW_val(1); 

%------------Same specifications given in JART------------%
PbRip_dB    = 0.05;                % 20log (1+PbRip) dB
SbRip_dB    = 80;                  % 20log (SbRip) dB
PbRip       = 10^(PbRip_dB/20)-1;
SbRip       = 10^(-SbRip_dB/20);
%--------------------Fixed filter-------------------------% 
cf          = fixedBW/fs;
pb          = cf;  
del         = 2600/fs; 
sb          = pb+del;
[n,fo,mo,w] = firpmord( [pb sb], [1 0], [PbRip,SbRip], 1);
hh          = firpm(n,fo,mo,w);
[h1,w]      = freqz (hh);

% %--------------- Profound loss Right Ear ------------------% 0.8867
[x2,y2,audiogram_type] = get_audiogram(35);

BW     = [200     200     300      300       300      400];   %2600
shift1 = [0.025     0.1248   0.2    0.28     0.36    0.48]; 
gain   = [86.9     86.5    91     85      89     93.7]; 

 audiogram_matching(BW,BW_val,fixedBW,shift1,gain,hh,x2,y2,audiogram_type);
